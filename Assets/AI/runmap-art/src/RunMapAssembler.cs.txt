using System;
using System.Collections.Generic;
using UnityEditor;
using UnityEngine;
using UnityEngine.Tilemaps;
using RunMap;

namespace RunMapTool.Editor
{
    /// <summary>
    /// **Pass E（装配）**：把 <see cref="MapLayout"/> 变成场景里的实体。
    ///
    /// 两件事：
    /// ① 按房间格坐标实例化房间块 prefab（1 格 = 40×24 世界单位）；
    /// ② **封堵未使用的开口**。
    ///
    /// ⚠ 为什么封堵是必须的（不是可选优化）：
    /// v2 规格让每个房间都带 L/R/U/D 四向开口（否则支路没法在 2D 网格里展开）。
    /// 那些"没接东西的洞口"如果留着，玩家会从天花板 / 地板掉出地图。
    /// 封堵规则很简单 —— 把该方向那条边界带补成实心瓦片：
    ///   · 左右：列 0..1 / W-2..W-1，行 3..H-4（墙体带；开了门的那段会被补回成墙）
    ///   · 上下：列 19..21，行 0..2（地板带）/ H-2..H-1（天花板带）
    /// 已经实心的格子不会被动（封堵前先判空），所以对 Entry（本来就左墙封死）和竖井块都是幂等的。
    /// </summary>
    public static class RunMapAssembler
    {
        public static GameObject Assemble(MapLayout layout, RunMapCatalog cat, string prefabRoot, string tileRoot, Transform parent)
        {
            Dictionary<ushort, List<SocketDir>> used = CollectUsed(layout);
            var root = new GameObject(layout.ThemeKey + "_runmap_" + layout.Seed);
            if (parent != null) root.transform.SetParent(parent, false);

            var fillCache = new Dictionary<string, Tile>();
            var topCache = new Dictionary<string, Tile>();
            int sealedTotal = 0, placed = 0;
            Decorations.Clear();

            for (int i = 0; i < layout.Nodes.Count; i++)
            {
                RoomNode n = layout.Nodes[i];
                string theme = cat.ThemeKeyForLayer(n.Layer);
                string typeName = TypeName(n.Kind);
                string path = prefabRoot + "/" + theme + "/Chunk_" + theme + "_" + typeName + ".prefab";
                GameObject prefab = AssetDatabase.LoadAssetAtPath<GameObject>(path);
                if (prefab == null)
                {
                    Debug.LogError("[RunMapAssembler] 缺 prefab：" + path);
                    continue;
                }

                GameObject go = (GameObject)PrefabUtility.InstantiatePrefab(prefab);
                go.transform.SetParent(root.transform, false);
                go.transform.localPosition = new Vector3(n.CellX * RunMapConst.ChunkW, n.CellY * RunMapConst.ChunkH, 0f);
                go.name = "R" + n.Id + "_" + typeName + "_L" + (int)n.Layer;
                placed++;

                Tile fill;
                if (!fillCache.TryGetValue(theme, out fill))
                {
                    fill = AssetDatabase.LoadAssetAtPath<Tile>(tileRoot + "/" + theme + "/Tile_Fill.asset");
                    fillCache[theme] = fill;
                }
                if (fill == null) continue;

                Tile top;
                if (!topCache.TryGetValue(theme, out top))
                {
                    top = AssetDatabase.LoadAssetAtPath<Tile>(tileRoot + "/" + theme + "/Tile_Top.asset");
                    topCache[theme] = top;
                }

                List<SocketDir> dirs = used.ContainsKey(n.Id) ? used[n.Id] : new List<SocketDir>();
                sealedTotal += SealUnused(go, fill, dirs);

                // Pass E 的最后一步：给房间加"地形起伏"（确定性，见 Decorate 的注释）
                string deco = Decorate(go, n, layout, fill, top);
                if (deco != "none") Decorations.Add("R" + n.Id + "(" + n.Kind + ")=" + deco);
            }

            Debug.Log("[RunMapAssembler] 房间 " + placed + " 个，封堵格 " + sealedTotal + " 个，地形修饰 " + Decorations.Count + " 处（主题 " + layout.ThemeKey + "，种子 " + layout.Seed + "）");
            return root;
        }

        /// <summary>把一条边界带上"空的格子"补成实心（已实心的不动，所以幂等）。</summary>
        private static int SealUnused(GameObject chunk, Tile fill, List<SocketDir> usedDirs)
        {
            Transform t = chunk.transform.Find("Tilemap_Ground");
            if (t == null) return 0;
            Tilemap tm = t.GetComponent<Tilemap>();
            if (tm == null) return 0;

            int n = 0;
            if (!usedDirs.Contains(SocketDir.Left))
                n += FillBand(tm, fill, 0, 1, 3, RunMapConst.ChunkH - 4);
            if (!usedDirs.Contains(SocketDir.Right))
                n += FillBand(tm, fill, RunMapConst.ChunkW - 2, RunMapConst.ChunkW - 1, 3, RunMapConst.ChunkH - 4);
            if (!usedDirs.Contains(SocketDir.Up))
                n += FillBand(tm, fill, RunMapConst.ShaftColumn - 1, RunMapConst.ShaftColumn + 1, RunMapConst.ChunkH - 2, RunMapConst.ChunkH - 1);
            if (!usedDirs.Contains(SocketDir.Down))
                n += FillBand(tm, fill, RunMapConst.ShaftColumn - 1, RunMapConst.ShaftColumn + 1, 0, 2);

            if (n > 0) tm.RefreshAllTiles();
            return n;
        }

        private static int FillBand(Tilemap tm, Tile fill, int x0, int x1, int y0, int y1)
        {
            int n = 0;
            for (int x = x0; x <= x1; x++)
                for (int y = y0; y <= y1; y++)
                {
                    var cell = new Vector3Int(x, y, 0);
                    if (tm.GetTile(cell) != null) continue;      // 已经实心 → 不动
                    tm.SetTile(cell, fill);
                    n++;
                }
            return n;
        }

        /// <summary>每个房间实际用到的接口方向（由边推出来）。</summary>
        public static Dictionary<ushort, List<SocketDir>> CollectUsed(MapLayout layout)
        {
            var used = new Dictionary<ushort, List<SocketDir>>();
            for (int i = 0; i < layout.Edges.Count; i++)
            {
                RoomEdge e = layout.Edges[i];
                Add(used, e.From, e.Dir);
                Add(used, e.To, RunMapConst.Opposite(e.Dir));
            }
            return used;
        }

        private static void Add(Dictionary<ushort, List<SocketDir>> map, ushort id, SocketDir dir)
        {
            List<SocketDir> list;
            if (!map.TryGetValue(id, out list)) { list = new List<SocketDir>(); map[id] = list; }
            if (!list.Contains(dir)) list.Add(dir);
        }

        // ═══════════════════════════════════════════════════════════
        //  地形起伏（Pass E 的最后一步）
        // ═══════════════════════════════════════════════════════════

        /// <summary>最近一次装配里每个房间用了什么地形修饰（给校验工具看）。</summary>
        public static readonly List<string> Decorations = new List<string>();

        /// <summary>
        /// 按"场景种子 + 房间 ID"确定性地给房间加地形起伏：高台 / 坑 / 柱子 / 台阶。
        ///
        /// **为什么不给每种房间做 prefab 变体**：6 型 × N 变体 × 3 主题会爆；
        /// 而"在规范块上做加法"既不动接口契约（门洞高度不变），又能让每局的每个房间长得不一样。
        ///
        /// **硬约束**（破了就会搞坏连通性或堵门）：
        ///   · 外侧 4 列不动 —— 那是门口通道；
        ///   · 竖井列 18..22 不动 —— 那是上下开口的通道；
        ///   · 坑最多挖掉第 1、2 行，**第 0 行永远保留**（有底、跳得出来）；
        ///   · 高台 / 柱子不超过 4 格高（跳得上去）。
        /// 只作用于 Combat / Corridor：入口、死路、竖井、boss 门的形状有语义，不该被随机改。
        /// </summary>
        private static string Decorate(GameObject chunk, RoomNode node, MapLayout layout, Tile fill, Tile top)
        {
            if (fill == null) return "none";
            if (node.Kind != ChunkKind.Combat && node.Kind != ChunkKind.Corridor) return "none";

            Transform t = chunk.transform.Find("Tilemap_Ground");
            if (t == null) return "none";
            Tilemap tm = t.GetComponent<Tilemap>();
            if (tm == null) return "none";

            const int Margin = 4;                                  // 外侧留白
            int lo = Margin, hi = RunMapConst.ChunkW - Margin - 1;  // 可动列范围
            var rng = new RunMapRng(layout.Seed * 31u + node.Id * 2654435761u + 7u);
            int pick = rng.Range(4);
            int n = 0;

            switch (pick)
            {
                case 0: // 高台：一段抬高的平台（顶上能站）
                {
                    int w = rng.Range(6, 11);
                    int h = rng.Range(2, 4);
                    int x0 = rng.Range(lo, Math.Max(lo, hi - w));
                    if (Blocked(x0, w)) return "none";
                    for (int x = x0; x < x0 + w; x++)
                        for (int y = 3; y <= 2 + h; y++)
                        {
                            tm.SetTile(new Vector3Int(x, y, 0), y == 2 + h ? top : fill);
                            n++;
                        }
                    tm.RefreshAllTiles();
                    return "terrace@x" + x0 + "w" + w + "h" + h;
                }
                case 1: // 坑：挖掉第 1、2 行（第 0 行留底）
                {
                    int w = rng.Range(5, 9);
                    int x0 = rng.Range(lo, Math.Max(lo, hi - w));
                    if (Blocked(x0, w)) return "none";
                    for (int x = x0; x < x0 + w; x++)
                        for (int y = 1; y <= 2; y++)
                        {
                            if (tm.GetTile(new Vector3Int(x, y, 0)) == null) continue;
                            tm.SetTile(new Vector3Int(x, y, 0), null);
                            n++;
                        }
                    tm.RefreshAllTiles();
                    return "pit@x" + x0 + "w" + w;
                }
                case 2: // 柱子：2~3 根
                {
                    int count = rng.Range(2, 3);
                    var xs = new List<int>();
                    for (int i = 0; i < count; i++)
                    {
                        int x = rng.Range(lo, hi);
                        if (Blocked(x, 1) || xs.Contains(x) || xs.Contains(x - 1) || xs.Contains(x + 1)) continue;
                        int h = rng.Range(3, 5);
                        for (int y = 3; y <= 2 + h; y++) { tm.SetTile(new Vector3Int(x, y, 0), fill); n++; }
                        xs.Add(x);
                    }
                    if (n == 0) return "none";
                    tm.RefreshAllTiles();
                    return "pillars×" + xs.Count;
                }
                default: // 台阶：4 级递升
                {
                    int x0 = rng.Range(lo, Math.Max(lo, hi - 16));
                    if (Blocked(x0, 16)) return "none";
                    for (int s = 0; s < 4; s++)
                        for (int k = 0; k <= s; k++)
                        {
                            int x = x0 + s * 4 + k;
                            int y = 3 + k;
                            if (k == s) tm.SetTile(new Vector3Int(x, y - 1, 0), top);
                            else tm.SetTile(new Vector3Int(x, y - 1, 0), fill);
                            n++;
                        }
                    tm.RefreshAllTiles();
                    return "steps@x" + x0;
                }
            }
        }

        /// <summary>这段列是否压到竖井通道（压到就别放地形，不然上下通路被堵）。</summary>
        private static bool Blocked(int x0, int w)
        {
            int a = x0, b = x0 + w - 1;
            return !(b < RunMapConst.ShaftColumn - 2 || a > RunMapConst.ShaftColumn + 2);
        }

        public static string TypeName(ChunkKind kind)
        {
            switch (kind)
            {
                case ChunkKind.Entry: return "Entry";
                case ChunkKind.Corridor: return "Corridor";
                case ChunkKind.Combat: return "Combat";
                case ChunkKind.DeadEnd: return "DeadEnd";
                case ChunkKind.LayerLink: return "LayerLink_Stair";
                case ChunkKind.BossGate: return "BossGate";
                default: return kind.ToString();
            }
        }
    }
}
